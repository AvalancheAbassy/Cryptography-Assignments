def intshift(X,a):
    K = ZZ^len(X)
    V = K(0)
    for i in range(len(X)):
        V[i] = X[(i+a)% len(X)]
    return V
def numberofentries(N,a):
    t = 0
    for i in range(len(N)):
        if N[i] == a:
            t = t+1
    return t
def findlength(N,T):
    K = ZZ^T
    V = K(0)
    for i in range(T):
        V[i] = numberofentries(N - intshift(N,i),0)
    return V
def realshift(X,a):
    K = RR^len(X)
    V = K(0)
    for i in range(len(X)):
        V[i] = X[(i-a)% len(X)]
    return V
def specialvector(X,l,a):
    K = ZZ^(floor(len(X)/l))
    W = K(0)
    for i in range(floor(len(X)/l)):
        W[i] = X[l*i+a]
    return W
def Wvector(X,l,a):
    K = RR^26
    W = K(0)
    for i in range(26):
        W[i] = numberofentries(specialvector(X,l,a),i)/(floor(len(X)/l))
    return W

    K = ZZ^len(N)
    P = K(0)
    for a in range(l):
        for i in range(floor(len(N)/l)):

    return P
